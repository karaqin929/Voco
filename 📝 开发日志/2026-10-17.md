- **历史补录**（用户已手动执行，success）：

```sql
-- 首版 r.content::jsonb 强转报错（22P02：库里弯引号老日报不是严格 JSON）
-- 终版：不解析整篇 JSON，正则直接抠 topic 值（直/弯引号均认），免疫弯引号
insert into topics (user_id, title, practice_count, last_practiced_at, created_at)
select t.user_id, t.topic, count(*), now(), now()
from (
  select r.user_id,
         substring(r.content from '["“]topic["”][[:space:]]*:[[:space:]]*["“]([^"”]*)["”]') as topic
  from reports r
  where r.content ~ '^\s*\{'
) t
where t.topic is not null and t.topic <> ''
  and not exists (
    select 1 from topics tp where tp.user_id = t.user_id and tp.title = t.topic
  )
group by t.user_id, t.topic;
```
